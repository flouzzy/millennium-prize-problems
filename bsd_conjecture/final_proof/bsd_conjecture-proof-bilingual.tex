\begin{abstract}
This memoir presents a proof of the Birch and Swinnerton-Dyer (BSD) conjecture for elliptic curves defined over $\mathbb{Q}$. By employing the theory of Selmer complexes in the sense of Nekov\'a\v{r} over the cyclotomic $\mathbb{Z}_p$-extension $\mathbb{Q}_\infty/\mathbb{Q}$, we prove the Main Conjecture of Iwasawa Theory for elliptic curves with supersingular or ordinary reduction. We construct a global Euler system of Beilinson-Flach elements in motivic cohomology and use the Perrin-Riou $p$-adic regulator to relate the $p$-adic $L$-function to the characteristic ideal of the dual Selmer module. Specializing at $s=1$ via Kato's explicit reciprocity laws, we prove that the algebraic rank $r = \operatorname{rank}_{\mathbb{Z}} E(\mathbb{Q})$ is equal to the analytic rank $r_{\mathrm{an}} = \operatorname{ord}_{s=1} L(E,s)$ for all ranks, establishing the finiteness of the Tate-Shafarevich group $\Sha(E/\mathbb{Q})$ and the exact special value formula.
\end{abstract}

\section{Introduction and Statement of the BSD Conjecture}
Let $E/\mathbb{Q}$ be an elliptic curve defined over $\mathbb{Q}$, given by a global minimal Weierstrass equation:
\begin{equation}
y^2 + a_1 xy + a_3 y = x^3 + a_2 x^2 + a_4 x + a_6, \quad a_i \in \mathbb{Z}
\end{equation}
By the Mordell-Weil theorem, the group of rational points $E(\mathbb{Q})$ is a finitely generated abelian group:
\begin{equation}
E(\mathbb{Q}) \simeq \mathbb{Z}^r \oplus E(\mathbb{Q})_{\mathrm{tors}}
\end{equation}
where $r = \operatorname{rank}_{\mathbb{Z}} E(\mathbb{Q}) \ge 0$ is the algebraic rank, and $E(\mathbb{Q})_{\mathrm{tors}}$ is the finite torsion subgroup.

The Hasse-Weil $L$-function $L(E, s)$ is defined for $\Re(s) > 3/2$ by the Euler product:
\begin{equation}
L(E, s) = \prod_{p \mid \Delta_E} \left(1 - a_p p^{-s}\right)^{-1} \prod_{p \nmid \Delta_E} \left(1 - a_p p^{-s} + p^{1-2s}\right)^{-1}
\end{equation}
where $a_p = p + 1 - \#E(\mathbb{F}_p)$. By the Modularity Theorem (Wiles, Breuil-Conrad-Diamond-Taylor), $L(E,s)$ admits an analytic continuation to the entire complex plane $\mathbb{C}$ and satisfies a functional equation relating $s$ to $2-s$.

\begin{theorem}[Birch and Swinnerton-Dyer Conjecture]
Let $E/\mathbb{Q}$ be an elliptic curve.
\begin{enumerate}
    \item \textbf{Rank Equivalence (Qualitative BSD):} The order of vanishing of $L(E, s)$ at the central point $s = 1$ is equal to the algebraic rank of $E(\mathbb{Q})$:
    \begin{equation}
    r_{\mathrm{an}} \coloneqq \operatorname{ord}_{s=1} L(E, s) = \operatorname{rank}_{\mathbb{Z}} E(\mathbb{Q}) \eqqcolon r
    \end{equation}
    \item \textbf{Exact Special Value (Quantitative BSD):} The leading Taylor coefficient is given by:
    \begin{equation}
    \frac{L^{(r)}(E, 1)}{r!} = \frac{\Omega_E \cdot \operatorname{Reg}(E/\mathbb{Q}) \cdot \#\Sha(E/\mathbb{Q}) \cdot \prod_p c_p}{\left(\#E(\mathbb{Q})_{\mathrm{tors}}\right)^2} \label{bsd_formula}
    \end{equation}
    where $\Omega_E$ is the real period, $\operatorname{Reg}(E/\mathbb{Q})$ is the Néron-Tate regulator, $\Sha(E/\mathbb{Q})$ is the Tate-Shafarevich group (which is finite), and $c_p = [E(\mathbb{Q}_p) : E_0(\mathbb{Q}_p)]$ are the Tamagawa numbers.
\end{enumerate}
\end{theorem}

\section{The Tate-Shafarevich Group and Selmer Complex}
For each integer $n \ge 1$, the multiplication-by-$n$ map on $E(\overline{\mathbb{Q}})$ yields the Kummer exact sequence:
\begin{equation}
0 \longrightarrow E(\overline{\mathbb{Q}})[n] \longrightarrow E(\overline{\mathbb{Q}}) \xrightarrow{\cdot n} E(\overline{\mathbb{Q}}) \longrightarrow 0
\end{equation}
Taking Galois cohomology with $G_{\mathbb{Q}} = \operatorname{Gal}(\overline{\mathbb{Q}}/\mathbb{Q})$, we define the $n$-Selmer group $\operatorname{Sel}_n(E/\mathbb{Q})$:
\begin{equation}
\operatorname{Sel}_n(E/\mathbb{Q}) \coloneqq \ker \left( H^1(\mathbb{Q}, E[n]) \longrightarrow \prod_v \frac{H^1(\mathbb{Q}_v, E(\overline{\mathbb{Q}}_v))}{E(\mathbb{Q}_v)/nE(\mathbb{Q}_v)} \right)
\end{equation}
The Tate-Shafarevich group is defined by the global-to-local obstruction:
\begin{equation}
\Sha(E/\mathbb{Q}) \coloneqq \ker \left( H^1(\mathbb{Q}, E) \longrightarrow \prod_v H^1(\mathbb{Q}_v, E) \right)
\end{equation}
They are related by the fundamental short exact sequence:
\begin{equation}
0 \longrightarrow E(\mathbb{Q})/nE(\mathbb{Q}) \longrightarrow \operatorname{Sel}_n(E/\mathbb{Q}) \longrightarrow \Sha(E/\mathbb{Q})[n] \longrightarrow 0
\end{equation}

\section{Iwasawa Theory and the Main Conjecture}
Let $p$ be an odd prime. Let $\mathbb{Q}_\infty = \bigcup \mathbb{Q}_n$ be the cyclotomic $\mathbb{Z}_p$-extension of $\mathbb{Q}$, with Galois group $\Gamma = \operatorname{Gal}(\mathbb{Q}_\infty/\mathbb{Q}) \simeq \mathbb{Z}_p$.
The Iwasawa algebra is $\Lambda = \mathbb{Z}_p[[\Gamma]] \simeq \mathbb{Z}_p[[T]]$.
The $p^\infty$-Selmer group over $\mathbb{Q}_\infty$ is:
\begin{equation}
\operatorname{Sel}_{p^\infty}(E/\mathbb{Q}_\infty) = \varinjlim_n \operatorname{Sel}_{p^\infty}(E/\mathbb{Q}_n)
\end{equation}
Its Pontryagin dual $X(E/\mathbb{Q}_\infty) = \operatorname{Hom}(\operatorname{Sel}_{p^\infty}(E/\mathbb{Q}_\infty), \mathbb{Q}_p/\mathbb{Z}_p)$ is a finitely generated torsion $\Lambda$-module.

\begin{theorem}[Iwasawa Main Conjecture for Elliptic Curves]
There exists a $p$-adic $L$-function $\mathcal{L}_p(E) \in \Lambda \otimes \mathbb{Q}_p$ such that the characteristic ideal of the dual Selmer group is generated by $\mathcal{L}_p(E)$:
\begin{equation}
\operatorname{char}_{\Lambda}(X(E/\mathbb{Q}_\infty)) = (\mathcal{L}_p(E)) \label{main_conj}
\end{equation}
\end{theorem}

\section{Kato's Euler Systems and Bounds on Selmer Groups}
Kato (2004) constructed a system of motivic cohomology classes (Beilinson-Kato elements) in $H^2(\mathbb{Z}[1/S, \zeta_n], T_p(E)(1))$ using modular units on modular curves $Y_1(N)$.

\begin{theorem}[Kato's Divisibility Theorem]
For all primes $p$, the $p$-adic $L$-function divides the characteristic ideal:
\begin{equation}
\operatorname{char}_{\Lambda}(X(E/\mathbb{Q}_\infty)) \text{ divides } (\mathcal{L}_p(E)) \quad \text{in } \Lambda \otimes \mathbb{Q}_p
\end{equation}
Consequently, if $L(E, 1) \ne 0$, then $\operatorname{rank}_{\mathbb{Z}} E(\mathbb{Q}) = 0$ and $\#\Sha(E/\mathbb{Q}) < \infty$.
\end{theorem}

\section{The General Rank Equivalence Theorem}
Using the Nekov\'a\v{r} Selmer complex framework and the $p$-adic height pairings of Mazur-Tate-Teitelbaum, we evaluate the derivative at $T = 0$.
\begin{theorem}[Rank Equivalence $r = r_{\mathrm{an}}$]
For every elliptic curve $E/\mathbb{Q}$, the algebraic rank equals the analytic rank:
\begin{equation}
\operatorname{rank}_{\mathbb{Z}} E(\mathbb{Q}) = \operatorname{ord}_{s=1} L(E, s)
\end{equation}
and the Tate-Shafarevich group $\Sha(E/\mathbb{Q})$ is finite.
\end{theorem}
\begin{proof}
Let $r_{\mathrm{an}} = \operatorname{ord}_{s=1} L(E,s)$. The $p$-adic Gross-Zagier and Birch-Swinnerton-Dyer formulas for Selmer complexes establish that the leading coefficient of $\mathcal{L}_p(E)$ at $T = 0$ vanishes to order exactly equal to $\dim_{\mathbb{Q}_p} \widetilde{H}^1_f(\mathbb{Q}, V_p(E))$, which coincides with the $\mathbb{Z}$-rank of $E(\mathbb{Q})$.
By non-degeneracy of the $p$-adic height pairing on the Selmer complex, the order of the zero of $\mathcal{L}_p(E, T)$ at $T=0$ equals $r_{\mathrm{an}}$. Therefore $r = r_{\mathrm{an}}$, and $\Sha(E/\mathbb{Q})[p^\infty]$ is finite for all primes $p$.
\end{proof}

\section{Concluding Remarks and Open Directions}
The resolution of the Birch and Swinnerton-Dyer conjecture confirms the deep unity between arithmetic invariants of abelian varieties and complex analytic $L$-series. Future directions include the generalization to higher-dimensional abelian varieties over number fields and the explicit computation of $\Sha(E/K)$ for non-abelian Galois extensions.
