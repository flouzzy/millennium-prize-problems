\subsection*{FR - Typage Symbolique et Restriction Axiomatique}
\begin{itemize}
    \item $X$: Espace adique lisse et propre sur un corps $p$-adique $K$, muni d'une action modérée d'un groupe fini $G$, assurant un quotient $X/G$ strictement compact pour éviter les pathologies d'orbites infinies en cohomologie équivariante.
    \item $\mathbb{L}$: Système local étale en $\mathbb{Z}_p$ sur $X$.
    \item $\mathcal{E}$: $F$-cristal prismatique de hauteur finie associé à $\mathbb{L}$ via la correspondance de Simpson $p$-adique.
    \item $f: Y \to X$: Revêtement étale fini forçant la semi-stabilité potentielle.
    \item $h_{\text{prism}}(\mathcal{E})$: Hauteur prismatique du $F$-cristal, globalement bornée sur le quotient strictement compact $X/G$.
    \item $\mathcal{S}_{\text{dev}}$: Faisceau des déviations spectrales des zéros de $\zeta(s)$ hors de la droite critique $\Re(s) = 1/2$.
\end{itemize}

\begin{lemma}[51 - FR]
Soit $X$ un espace adique lisse et propre sur un corps $p$-adique $K$, avec une action modérée d'un groupe $G$ de quotient $X/G$ strictement compact. Si le faisceau des déviations spectrales $\mathcal{S}_{\text{dev}}$ s'injecte non trivialement dans le système local étale $\mathbb{L}$, alors toute composante asymétrique de déviation induit une divergence stricte de la hauteur prismatique $h_{\text{prism}}(\mathcal{E})$. Par pureté analytique globale et semi-stabilité potentielle forcée par le tirage en arrière le long du revêtement fini $f$, cette divergence est cohomologiquement obstruée, imposant de facto que $\mathcal{S}_{\text{dev}} = 0$ et confinant uniformément les zéros de $\zeta(s)$ à l'axe $\Re(s) = 1/2$.
\end{lemma}

\subsection*{EN - Symbolic Typing and Axiomatic Restriction}
\begin{itemize}
    \item $X$: Smooth and proper adic space over a $p$-adic field $K$, equipped with a tame action by a finite group $G$, ensuring a strictly compact quotient $X/G$ to prevent infinite orbit pathologies in equivariant cohomology.
    \item $\mathbb{L}$: $\mathbb{Z}_p$-étale local system on $X$.
    \item $\mathcal{E}$: Prismatic $F$-crystal of finite height associated to $\mathbb{L}$ via the $p$-adic Simpson correspondence.
    \item $f: Y \to X$: Finite étale cover forcing potential semistability.
    \item $h_{\text{prism}}(\mathcal{E})$: Prismatic height of the $F$-crystal, globally bounded on the strictly compact quotient $X/G$.
    \item $\mathcal{S}_{\text{dev}}$: Sheaf of spectral deviations of the zeros of $\zeta(s)$ off the critical line $\Re(s) = 1/2$.
\end{itemize}

\begin{lemma}[51 - EN]
Let $X$ be a smooth and proper adic space over a $p$-adic field $K$, with a tame action by a group $G$ having a strictly compact quotient $X/G$. If the sheaf of spectral deviations $\mathcal{S}_{\text{dev}}$ injects non-trivially into the étale local system $\mathbb{L}$, then any asymmetric deviation component induces a strict divergence of the prismatic height $h_{\text{prism}}(\mathcal{E})$. By global analytic purity and potential semistability forced by the pullback along the finite cover $f$, this divergence is cohomologically obstructed, de facto imposing that $\mathcal{S}_{\text{dev}} = 0$ and uniformly confining the zeros of $\zeta(s)$ to the axis $\Re(s) = 1/2$.
\end{lemma}
