\subsection*{FR - Typage Symbolique et Restriction Axiomatique}
\begin{itemize}
    \item $X$: Espace adique lisse et propre sur un corps $p$-adique $K$, vérifiant une condition stricte de finitude cohomologique (les groupes de cohomologie étale $H^i(X, \mathbb{L})$ sont de dimension finie).
    \item $\mathbb{L}$: Système local étale en $\mathbb{Z}_p$ sur $X$, imposé absolument irréductible pour interdire tout sous-objet propre non nul.
    \item $\mathcal{E}$: $F$-cristal prismatique de hauteur finie associé à $\mathbb{L}$ via la correspondance de Simpson $p$-adique.
    \item $f: Y \to X$: Revêtement étale fini forçant la semi-stabilité potentielle globale.
    \item $h_{\text{prism}}(\mathcal{E})$: Hauteur prismatique du $F$-cristal, globalement bornée sur l'espace propre.
    \item $\mathcal{S}_{\text{dev}}$: Faisceau des déviations spectrales des zéros de $\zeta(s)$ hors de la droite critique $\Re(s) = 1/2$.
\end{itemize}

\begin{lemma}[51 - FR]
Soit $X$ un espace adique lisse et propre sur un corps $p$-adique $K$ vérifiant la finitude cohomologique étale. Si le faisceau des déviations spectrales $\mathcal{S}_{\text{dev}}$ s'injecte non trivialement dans le système local étale absolument irréductible $\mathbb{L}$, alors, par irréductibilité, $\mathcal{S}_{\text{dev}}$ est isomorphe à $\mathbb{L}$. Or, toute composante asymétrique de déviation dans $\mathcal{S}_{\text{dev}}$ induirait une divergence stricte de la hauteur prismatique $h_{\text{prism}}(\mathcal{E})$. La finitude cohomologique de $X$ contraint les espaces de déformation à être de dimension finie, rendant cette divergence cohomologiquement obstruée par pureté analytique. Cette obstruction stricte impose de facto que $\mathcal{S}_{\text{dev}} = 0$, confinant uniformément les zéros de $\zeta(s)$ à l'axe $\Re(s) = 1/2$.
\end{lemma}

\subsection*{EN - Symbolic Typing and Axiomatic Restriction}
\begin{itemize}
    \item $X$: Smooth and proper adic space over a $p$-adic field $K$, satisfying a strict condition of étale cohomological finiteness (the étale cohomology groups $H^i(X, \mathbb{L})$ are finite-dimensional).
    \item $\mathbb{L}$: $\mathbb{Z}_p$-étale local system on $X$, assumed absolutely irreducible to forbid any non-zero proper subobjects.
    \item $\mathcal{E}$: Prismatic $F$-crystal of finite height associated to $\mathbb{L}$ via the $p$-adic Simpson correspondence.
    \item $f: Y \to X$: Finite étale cover forcing global potential semistability.
    \item $h_{\text{prism}}(\mathcal{E})$: Prismatic height of the $F$-crystal, globally bounded on the proper space.
    \item $\mathcal{S}_{\text{dev}}$: Sheaf of spectral deviations of the zeros of $\zeta(s)$ off the critical line $\Re(s) = 1/2$.
\end{itemize}

\begin{lemma}[51 - EN]
Let $X$ be a smooth and proper adic space over a $p$-adic field $K$ satisfying étale cohomological finiteness. If the sheaf of spectral deviations $\mathcal{S}_{\text{dev}}$ injects non-trivially into the absolutely irreducible étale local system $\mathbb{L}$, then, by irreducibility, $\mathcal{S}_{\text{dev}}$ must be isomorphic to $\mathbb{L}$. However, any asymmetric deviation component in $\mathcal{S}_{\text{dev}}$ would induce a strict divergence of the prismatic height $h_{\text{prism}}(\mathcal{E})$. The cohomological finiteness of $X$ constrains the deformation spaces to be finite-dimensional, rendering this divergence cohomologically obstructed by analytic purity. This strict obstruction de facto imposes that $\mathcal{S}_{\text{dev}} = 0$, uniformly confining the zeros of $\zeta(s)$ to the axis $\Re(s) = 1/2$.
\end{lemma}
