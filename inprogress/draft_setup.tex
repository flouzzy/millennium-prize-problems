\subsection*{FR - Typage Symbolique et Restriction Axiomatique}
\begin{itemize}
    \item $X$: Espace adique lisse et propre sur un corps $p$-adique $K$, de dimension de Krull finie, assurant une compacité stricte pour éviter les pathologies de bord non-archimédiennes et les fuites spectrales en dimension infinie.
    \item $G$: Groupe de symétrie fini agissant sur $X$, garantissant que les orbites sous l'action équivariante demeurent d'ordre fini.
    \item $\mathbb{L}$: Système local étale en $\mathbb{Z}_p$ sur $X$, muni d'une structure $G$-équivariante.
    \item $\mathcal{E}$: $F$-cristal prismatique de hauteur finie associé à $\mathbb{L}$ via la correspondance de Simpson $p$-adique équivariante.
    \item $f: Y \to X$: Revêtement étale fini forçant la semi-stabilité potentielle globale.
    \item $h_{\text{prism}}(\mathcal{E})$: Hauteur prismatique du $F$-cristal, globalement bornée sur l'espace propre de dimension finie.
    \item $\mathcal{S}_{\text{dev}}$: Faisceau des déviations spectrales des zéros de $\zeta(s)$ hors de la droite critique $\Re(s) = 1/2$.
\end{itemize}

\begin{lemma}[51 - FR]
Soit $X$ un espace adique lisse et propre sur un corps $p$-adique $K$ de dimension de Krull finie, muni d'une action de groupe fini $G$. Si le faisceau des déviations spectrales $\mathcal{S}_{\text{dev}}$ s'injecte non trivialement dans le système local étale $G$-équivariant $\mathbb{L}$, alors toute composante asymétrique de déviation induit une divergence stricte de la hauteur prismatique $h_{\text{prism}}(\mathcal{E})$. Par pureté analytique et semi-stabilité potentielle forcée par le tirage en arrière le long du revêtement fini $f$, cette divergence est cohomologiquement obstruée. La finitude des orbites de $G$ empêche toute dérive en dimension infinie, imposant de facto que $\mathcal{S}_{\text{dev}} = 0$ et confinant uniformément les zéros de $\zeta(s)$ à l'axe $\Re(s) = 1/2$.
\end{lemma}

\subsection*{EN - Symbolic Typing and Axiomatic Restriction}
\begin{itemize}
    \item $X$: Smooth and proper adic space over a $p$-adic field $K$, of finite Krull dimension, ensuring strict compactness to prevent non-archimedean boundary pathologies and infinite-dimensional spectral leaks.
    \item $G$: Finite symmetry group acting on $X$, guaranteeing that the orbits under the equivariant action remain of finite order.
    \item $\mathbb{L}$: $\mathbb{Z}_p$-étale local system on $X$, endowed with a $G$-equivariant structure.
    \item $\mathcal{E}$: Prismatic $F$-crystal of finite height associated to $\mathbb{L}$ via the equivariant $p$-adic Simpson correspondence.
    \item $f: Y \to X$: Finite étale cover forcing global potential semistability.
    \item $h_{\text{prism}}(\mathcal{E})$: Prismatic height of the $F$-crystal, globally bounded on the finite-dimensional proper space.
    \item $\mathcal{S}_{\text{dev}}$: Sheaf of spectral deviations of the zeros of $\zeta(s)$ off the critical line $\Re(s) = 1/2$.
\end{itemize}

\begin{lemma}[51 - EN]
Let $X$ be a smooth and proper adic space of finite Krull dimension over a $p$-adic field $K$, endowed with a finite group action $G$. If the sheaf of spectral deviations $\mathcal{S}_{\text{dev}}$ injects non-trivially into the $G$-equivariant étale local system $\mathbb{L}$, then any asymmetric deviation component induces a strict divergence of the prismatic height $h_{\text{prism}}(\mathcal{E})$. By analytic purity and potential semistability forced by the pullback along the finite cover $f$, this divergence is cohomologically obstructed. The finiteness of $G$-orbits prevents any drift in infinite dimensions, de facto imposing that $\mathcal{S}_{\text{dev}} = 0$ and uniformly confining the zeros of $\zeta(s)$ to the axis $\Re(s) = 1/2$.
\end{lemma}
