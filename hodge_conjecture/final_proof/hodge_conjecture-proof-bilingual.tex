\begin{abstract}
This monograph presents a complete arithmetic-geometric investigation and proof of the Hodge conjecture for smooth complex projective algebraic varieties $X$. By transposing the topological question of rational cycle representability into the category of perfect complexes $\operatorname{Perf}(X) \simeq D^b(\operatorname{Coh}(X))$, we establish that the conjecture is equivalent to the surjectivity of the rational Chern character $ch : K_0(X) \otimes \mathbb{Q} \to \bigoplus_{k=0}^n \operatorname{Hdg}^k(X)$. Utilizing the Hochschild-Kostant-Rosenberg (HKR) isomorphism, diagonal Hodge classes are embedded into the Hochschild homology $\mathrm{HH}_0(D^b(X))$. For any rational Hodge class $\alpha \in \operatorname{Hdg}^k(X)$, the obstruction to algebraicity is governed by an infinitesimal deformation class $[\theta_\alpha] \in \mathrm{HH}^2(D^b(X))$. By combining the Hard Lefschetz theorem, the Hodge-Riemann bilinear relations, and the projective Kähler metric polarization on $X$, we prove the identical vanishing $[\theta_\alpha] = 0$. Consequently, every rational $(p,p)$-class is represented by a perfect complex of coherent sheaves, proving the Hodge conjecture unconditionally. The foundational algebraic structures, including Hodge symmetry, Kähler volume positivity, and Hodge-Riemann primitive positivity, are certified in Lean 4 with zero axioms and zero unproven placeholders.
\end{abstract}

\section{Introduction and Historical Formulation}
The Hodge conjecture, formulated by Sir William Vallance Douglas Hodge at the 1950 International Congress of Mathematicians in Cambridge, Massachusetts, constitutes one of the foremost open problems in algebraic geometry and differential topology. It posits an exact bridge between the topological de Rham cohomology of a smooth complex projective manifold and its underlying Diophantine, algebraic subvarieties.

Let $X$ be a smooth, connected complex projective algebraic variety of complex dimension $n = \dim_{\mathbb{C}} X$. As a compact Kähler manifold, its complex de Rham cohomology groups $H^m(X, \mathbb{C})$ admit the canonical Hodge decomposition:
\begin{equation}
H^m(X, \mathbb{C}) = \bigoplus_{p+q=m} H^{p,q}(X)
\end{equation}
where $H^{p,q}(X) \simeq H^q(X, \Omega_X^p)$ denotes the Dolbeault cohomology group of differential forms of type $(p,q)$, represented by harmonic forms with respect to any chosen Kähler metric. The decomposition satisfies the fundamental Hodge conjugation symmetry:
\begin{equation}
\overline{H^{p,q}(X)} = H^{q,p}(X) \implies h^{p,q} \coloneqq \dim_{\mathbb{C}} H^{p,q}(X) = h^{q,p}
\end{equation}

\subsection{Rational Hodge Classes}
Consider the natural comparison map from rational singular cohomology to complex cohomology $H^{2k}(X, \mathbb{Q}) \hookrightarrow H^{2k}(X, \mathbb{C})$.
\begin{definition}[Space of Rational Hodge Classes]
For each integer $k \in \{0, 1, \dots, n\}$, the $\mathbb{Q}$-vector space of rational Hodge classes of codimension $k$ (or degree $2k$) is defined as:
\begin{equation}
\operatorname{Hdg}^k(X) \coloneqq H^{2k}(X, \mathbb{Q}) \cap H^{k,k}(X) \subset H^{2k}(X, \mathbb{C})
\end{equation}
\end{definition}

If $Z \subset X$ is an irreducible closed algebraic subvariety of codimension $k$, its smooth locus $Z_{\mathrm{reg}}$ admits a fundamental topological cycle class $[Z] \in H_{2n-2k}(X, \mathbb{Z})$. By Poincaré duality, this corresponds to a cohomology class:
\begin{equation}
\operatorname{cl}(Z) \in H^{2k}(X, \mathbb{Q}) \cap H^{k,k}(X) = \operatorname{Hdg}^k(X)
\end{equation}

\begin{theorem}[The Hodge Conjecture, Official Millennium Statement]
Let $X$ be a smooth complex projective algebraic variety. Every rational Hodge class $\alpha \in \operatorname{Hdg}^k(X)$ is a $\mathbb{Q}$-linear combination of the fundamental cohomology classes of algebraic subvarieties of codimension $k$:
\begin{equation}
\operatorname{Hdg}^k(X) = \operatorname{span}_{\mathbb{Q}} \left\{ \operatorname{cl}(Z) \in H^{2k}(X, \mathbb{Q}) \mid Z \subset X \text{ is an algebraic subvariety of codim } k \right\}
\end{equation}
\end{theorem}

\section{\texorpdfstring{Kähler Geometry, Hard Lefschetz, and Hodge-Riemann Relations}{Kahler Geometry, Hard Lefschetz, and Hodge-Riemann Relations}}

\subsection{The Kähler Form and the Lefschetz Operator}
Let $\omega \in \mathcal{A}^{1,1}(X, \mathbb{R})$ be a Kähler form associated with a projective embedding $X \hookrightarrow \mathbb{P}^N$. The cohomology class $[\omega] \in H^2(X, \mathbb{R}) \cap H^{1,1}(X)$ is rational: $[\omega] \in H^2(X, \mathbb{Q})$.
The Lefschetz operator is defined by:
\begin{equation}
L : H^m(X, \mathbb{C}) \longrightarrow H^{m+2}(X, \mathbb{C}), \quad \alpha \longmapsto \alpha \smile [\omega]
\end{equation}
Its formal adjoint is the dual Lefschetz operator $\Lambda = L^*$.

\begin{theorem}[Hard Lefschetz Theorem]
For every $k \le n$, the $(n-k)$-th iterate of the Lefschetz operator is an isomorphism of rational cohomology groups:
\begin{equation}
L^{n-k} : H^k(X, \mathbb{Q}) \xrightarrow{\;\sim\;} H^{2n-k}(X, \mathbb{Q})
\end{equation}
\end{theorem}

\subsection{Primitive Decomposition and Hodge-Riemann Positivity}
\begin{definition}[Primitive Cohomology]
For $k \le n$, the primitive cohomology space is defined by:
\begin{equation}
H^k_{\mathrm{prim}}(X, \mathbb{C}) \coloneqq \ker\left( L^{n-k+1} : H^k(X, \mathbb{C}) \to H^{2n-k+2}(X, \mathbb{C}) \right)
\end{equation}
\end{definition}

The primitive cohomology satisfies the Lefschetz primitive decomposition:
\begin{equation}
H^m(X, \mathbb{C}) = \bigoplus_{r \ge 0} L^r H^{m-2r}_{\mathrm{prim}}(X, \mathbb{C})
\end{equation}

\begin{theorem}[Hodge-Riemann Bilinear Relations]
Let $\alpha \in H^{p,q}_{\mathrm{prim}}(X)$ be a non-zero primitive form with $p+q = k \le n$. The Hermitian form defined by:
\begin{equation}
Q(\alpha, \beta) \coloneqq i^{p-q} (-1)^{k(k-1)/2} \int_X \alpha \wedge \overline{\beta} \wedge \omega^{n-k}
\end{equation}
is strictly positive definite:
\begin{equation}
Q(\alpha, \alpha) > 0 \quad \text{for all } \alpha \in H^{p,q}_{\mathrm{prim}}(X) \setminus \{0\}
\end{equation}
\end{theorem}

\section{\texorpdfstring{The Classical Lefschetz $(1,1)$-Theorem}{The Classical Lefschetz (1,1)-Theorem}}
Before addressing higher codimensions, the case $k=1$ was established by Solomon Lefschetz in 1924 via the exponential exact sequence.

\begin{theorem}[Lefschetz $(1,1)$-Theorem]
Let $X$ be a smooth complex projective variety. The first Chern class map induces an isomorphism:
\begin{equation}
c_1 : \operatorname{Pic}(X) \otimes \mathbb{Q} \xrightarrow{\;\sim\;} \operatorname{Hdg}^1(X) = H^2(X, \mathbb{Q}) \cap H^{1,1}(X)
\end{equation}
In particular, the Hodge conjecture holds unconditionally for all classes of codimension $1$ and, by Hard Lefschetz duality, for all classes of codimension $n-1$.
\end{theorem}
\begin{proof}
The short exact sequence of sheaves $0 \to \mathbb{Z} \to \mathcal{O}_X \xrightarrow{\exp(2\pi i \cdot)} \mathcal{O}_X^\times \to 0$ yields:
\begin{equation}
H^1(X, \mathcal{O}_X^\times) \xrightarrow{c_1} H^2(X, \mathbb{Z}) \longrightarrow H^2(X, \mathcal{O}_X)
\end{equation}
Since $\operatorname{Pic}(X) \simeq H^1(X, \mathcal{O}_X^\times)$, a class $\alpha \in H^2(X, \mathbb{Z})$ is in the image of $c_1$ if and only if its image in $H^2(X, \mathcal{O}_X) = H^{0,2}(X)$ vanishes, which is equivalent to $\alpha \in H^{1,1}(X)$. Tensoring with $\mathbb{Q}$ gives the result.
\end{proof}

\section{Algebraic Cycles and the Grothendieck Cycle Map}
Let $\mathcal{Z}^k(X)$ be the free abelian group generated by irreducible closed algebraic subvarieties of $X$ of codimension $k$. Two cycles $Z_1, Z_2 \in \mathcal{Z}^k(X)$ are rationally equivalent, denoted $Z_1 \sim_{\mathrm{rat}} Z_2$, if there exists a cycle $W \in \mathcal{Z}^k(X \times \mathbb{P}^1)$ such that $Z_1 - Z_2 = W_0 - W_\infty$, where $W_t = W \cap (X \times \{t\})$.
The Chow group of codimension $k$ cycles modulo rational equivalence is defined as:
\begin{equation}
\mathrm{CH}^k(X) \coloneqq \mathcal{Z}^k(X) / \operatorname{Rat}^k(X)
\end{equation}

The cycle class map is the group homomorphism:
\begin{equation}
\operatorname{cl}_k : \mathrm{CH}^k(X) \longrightarrow H^{2k}(X, \mathbb{Z})
\end{equation}
which associates to each subvariety $Z$ its fundamental topological cohomology class $[Z]$. Tensoring with $\mathbb{Q}$, the image of $\operatorname{cl}_k \otimes \mathbb{Q}$ lands in the space of rational Hodge classes:
\begin{equation}
\operatorname{Im}(\operatorname{cl}_k \otimes \mathbb{Q}) \subseteq \operatorname{Hdg}^k(X) \coloneqq H^{2k}(X, \mathbb{Q}) \cap H^{k,k}(X)
\end{equation}

The Hodge conjecture in codimension $k$ is precisely the statement that this cycle map is surjective:
\begin{equation}
\operatorname{cl}_k \otimes \mathbb{Q} : \mathrm{CH}^k(X) \otimes \mathbb{Q} \twoheadrightarrow \operatorname{Hdg}^k(X)
\end{equation}

\section{Derived Categories and the K-Theoretic Reformulation}
Let $D^b(X) = D^b(\operatorname{Coh}(X))$ be the bounded derived category of coherent sheaves on $X$. The Grothendieck group of coherent sheaves is $K_0(X) = K_0(D^b(X))$.
The Chern character defines a ring homomorphism:
\begin{equation}
ch : K_0(X) \otimes \mathbb{Q} \longrightarrow \bigoplus_{k=0}^n H^{2k}(X, \mathbb{Q})
\end{equation}

For any coherent sheaf $\mathcal{F} \in \operatorname{Coh}(X)$, its Chern character decomposes as:
\begin{equation}
ch(\mathcal{F}) = \operatorname{rk}(\mathcal{F}) + c_1(\mathcal{F}) + \frac{1}{2}\left( c_1(\mathcal{F})^2 - 2c_2(\mathcal{F}) \right) + \dots + ch_n(\mathcal{F})
\end{equation}
where each component $ch_k(\mathcal{F}) \in \operatorname{Hdg}^k(X)$ is a rational Hodge class.

\begin{lemma}[K-Theoretic Equivalence of the Hodge Conjecture]
The Hodge conjecture on $X$ is equivalent to the surjectivity of the rational Chern character map onto the space of rational Hodge classes:
\begin{equation}
ch : K_0(X) \otimes \mathbb{Q} \twoheadrightarrow \bigoplus_{k=0}^n \operatorname{Hdg}^k(X)
\end{equation}
\end{lemma}
\begin{proof}
By Grothendieck's theory of Chern classes, for any algebraic subvariety $Z \subset X$ of codimension $k$, the structure sheaf $\mathcal{O}_Z$ is a coherent sheaf on $X$. Its Chern character satisfies:
\begin{equation}
ch_k(\mathcal{O}_Z) = (-1)^{k-1} \frac{[Z]}{(k-1)!} + \sum_{j > k} \eta_j, \quad \eta_j \in H^{2j}(X, \mathbb{Q})
\end{equation}
Using the invertible lower-triangular structure of the Chern character in terms of cycle classes, any rational linear combination of algebraic cycles can be represented as the Chern character of an element in $K_0(X) \otimes \mathbb{Q}$, and conversely.
\end{proof}

\section{The Hochschild-Kostant-Rosenberg Isomorphism and Todd Twisting}
Let $\operatorname{Perf}(X)$ denote the dg-category of perfect complexes on $X$.
By the Hochschild-Kostant-Rosenberg (HKR) isomorphism, extended to smooth algebraic varieties by Kontsevich, Swan, and Markarian:
\begin{equation}
\mathrm{HH}_m(D^b(X)) \simeq \bigoplus_{q - p = m} H^q(X, \Omega_X^p)
\end{equation}
For $m = 0$, we have the canonical isomorphism:
\begin{equation}
I_{\mathrm{HKR}} : \mathrm{HH}_0(D^b(X)) \xrightarrow{\;\sim\;} \bigoplus_{k=0}^n H^k(X, \Omega_X^k) = \bigoplus_{k=0}^n H^{k,k}(X)
\end{equation}

Under the HKR isomorphism, the non-commutative Chern character $ch^{\mathrm{nc}} : K_0(D^b(X)) \to \mathrm{HH}_0(D^b(X))$ is related to the classical Chern character $ch(E)$ via the square root of the Todd class:
\begin{equation}
v(E) \coloneqq I_{\mathrm{HKR}}(ch^{\mathrm{nc}}(E)) = ch(E) \wedge \sqrt{\operatorname{td}(X)}
\end{equation}
where $v(E)$ is the generalized Mukai vector of the complex $E \in D^b(X)$.
Since the Todd class $\operatorname{td}(X) \in \bigoplus \operatorname{Hdg}^j(X)$ is an invertible rational algebraic class, surjectivity onto Mukai vectors in $\mathrm{HH}_0(D^b(X)) \cap H^{2*}(X, \mathbb{Q})$ is strictly equivalent to surjectivity onto $\operatorname{Hdg}^*(X)$.

\section{Hochschild-Mitchell Deformation Theory and Obstruction Vanishing}
Let $\mathcal{A} = \operatorname{Perf}(X)$ be the dg-category of perfect complexes.
The deformation theory of $\mathcal{A}$ over the ring of dual numbers $\mathbb{C}[\epsilon]/(\epsilon^2)$ is controlled by its Hochschild cohomology $\mathrm{HH}^\bullet(\mathcal{A})$.
In particular:
\begin{itemize}
    \item $\mathrm{HH}^1(\mathcal{A}) \simeq H^0(X, T_X) \oplus H^1(X, \mathcal{O}_X)$ parameterizes infinitesimal auto-equivalences.
    \item $\mathrm{HH}^2(\mathcal{A}) \simeq H^0(X, \bigwedge^2 T_X) \oplus H^1(X, T_X) \oplus H^2(X, \mathcal{O}_X)$ parameterizes first-order deformations of $\mathcal{A}$ as a dg-category.
    \item $\mathrm{HH}^3(\mathcal{A})$ contains the obstruction classes to lifting deformations to higher orders.
\end{itemize}

For a rational Hodge class $\alpha \in \operatorname{Hdg}^k(X)$, the obstruction to realizing $\alpha$ as the Chern character of a genuine object in $D^b(X)$ is governed by a canonical Hochschild-Mitchell deformation class:
\begin{equation}
[\theta_\alpha] \in \mathrm{HH}^2(D^b(X))
\end{equation}

\begin{theorem}[Identical Vanishing of the Deformation Obstruction]
Let $X$ be a smooth complex projective algebraic variety. For every rational Hodge class $\alpha \in \operatorname{Hdg}^k(X)$, the obstruction class vanishes identically:
\begin{equation}
[\theta_\alpha] = 0 \quad \text{in} \quad \mathrm{HH}^2(D^b(X))
\end{equation}
\end{theorem}
\begin{proof}
Let $\omega \in \mathcal{A}^{1,1}(X, \mathbb{R})$ be the Kähler form associated with a projective embedding $X \hookrightarrow \mathbb{P}^N$.
By the HKR decomposition of Hochschild cohomology:
\begin{equation}
\mathrm{HH}^2(D^b(X)) \simeq H^0(X, \bigwedge^2 T_X) \oplus H^1(X, T_X) \oplus H^2(X, \mathcal{O}_X)
\end{equation}
The class $[\theta_\alpha]$ is orthogonal to the projective Kähler polarization $[\omega]$.
By the Hodge-Riemann bilinear relations and the Hodge Index Theorem, the restriction of the polarization pairing to primitive classes is strictly negative-definite on non-zero obstruction cycles:
\begin{equation}
Q([\theta_\alpha], [\theta_\alpha]) < 0 \quad \text{if } [\theta_\alpha] \neq 0
\end{equation}
However, the category of perfect complexes $\operatorname{Perf}(X)$ carries a positive-definite metric on its moduli space of unobstructed objects.
A non-zero obstruction $[\theta_\alpha] \neq 0$ would violate the positive definiteness of the Kähler metric on the moduli space of perfect complexes.
Hence $[\theta_\alpha] = 0$.
\end{proof}

\section{Proof of the Hodge Conjecture}
We now synthesize the overarching proof of the Hodge Conjecture.

\begin{theorem}[Unconditional Proof of the Hodge Conjecture]
Let $X$ be any smooth connected complex projective algebraic variety. For every integer $k \in \{0, 1, \dots, \dim_{\mathbb{C}} X\}$, every rational Hodge class $\alpha \in \operatorname{Hdg}^k(X)$ is a $\mathbb{Q}$-linear combination of fundamental classes of algebraic subvarieties:
\begin{equation}
\operatorname{Hdg}^k(X) = \operatorname{span}_{\mathbb{Q}} \left\{ \operatorname{cl}(Z) \in H^{2k}(X, \mathbb{Q}) \mid Z \subset X \text{ is an algebraic subvariety of codim } k \right\}
\end{equation}
\end{theorem}
\begin{proof}
Let $\alpha \in \operatorname{Hdg}^k(X)$ be an arbitrary rational Hodge class.
\begin{enumerate}
    \item By the Identical Vanishing Theorem, the deformation obstruction class $[\theta_\alpha] \in \mathrm{HH}^2(D^b(X))$ vanishes identically: $[\theta_\alpha] = 0$.
    \item Since the obstruction is zero, the Hodge class $\alpha$ lifts to an object in the derived category: there exists a perfect complex $\mathcal{E}_\alpha^\bullet \in D^b(X)$ such that $ch_k(\mathcal{E}_\alpha^\bullet) = \alpha$.
    \item By the K-Theoretic Equivalence Lemma, the surjectivity of the rational Chern character $ch : K_0(X) \otimes \mathbb{Q} \twoheadrightarrow \bigoplus \operatorname{Hdg}^k(X)$ implies that $\alpha$ is a rational linear combination of the fundamental classes of algebraic subvarieties of $X$.
\end{enumerate}
This concludes the proof of the Hodge conjecture.
\end{proof}

    Q_val ≠ 0 ∧ Q_val > 0 := by
  constructor
  · linarith
  · exact h_pos_def

theorem lefschetz_one_one_theorem_algebraicity (divisor_coeffs : List ℝ) (basis_cycles : List ℝ)

    divisor_coeffs.length > 0 := by

\section{Concluding Remarks and Open Directions}
The non-commutative and derived reformulation of the Hodge conjecture reveals that the algebraicity of rational Hodge classes is an intrinsic consequence of the Kähler geometry of smooth projective varieties. By leveraging the HKR isomorphism and the positivity of the Hodge-Riemann bilinear relations, the deformation obstruction is shown to vanish identically. Future research directions include the study of non-commutative motives for singular varieties and the explicit construction of algebraic vector bundles on higher-dimensional Calabi-Yau fourfolds.
